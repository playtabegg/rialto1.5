# Rialto - Game Disc Builder
# Core dependencies
#
# Tested on Python 3.9 - 3.12 (64-bit) on Windows 10/11.
# Versions are intentionally range-pinned so a fresh clone (and the first-run
# `pip install -r requirements.txt` bootstrap in Rialto.pyw) installs a
# combination known to work across that range.
#
# --require-hashes is deliberately not used here. That mode needs a hash for
# every transitive wheel, and those wheels differ by Python minor version
# (3.9 vs 3.12 vs 3.14) and platform. A single hash-pinned file would break
# the 3.9-3.12 first-run bootstrap.

# Image processing
pillow>=10,<12

# GUI framework for the disc menu
PyQt5==5.15.11

# Windows-specific functionality (drive detection, shortcuts)
pywin32>=306

# Application packaging (rialto.spec / menu.spec builds, and the
# per-title menu fallback when no pre-built menu.exe is present)
pyinstaller>=6,<7
